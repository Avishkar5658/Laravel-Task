services:
  app:
    build: .
    container_name: cbs-laravel-app
    restart: unless-stopped
    working_dir: /var/www
    volumes:
      - ./src:/var/www
    env_file:
      - ./src/.env
    depends_on:
      db:
        condition: service_healthy
    networks:
      - cbs-internal

  web:
    image: nginx:alpine
    container_name: cbs-laravel-web
    restart: unless-stopped
    ports:
      - "9000:80"
    volumes:
      - ./src:/var/www:ro
      - ./docker/nginx/default.conf:/etc/nginx/conf.d/default.conf:ro
    depends_on:
      - app
    networks:
      - cbs-internal

  db:
    image: mysql:8.0
    container_name: cbs-laravel-db
    restart: unless-stopped
    environment:
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD:?Set MYSQL_ROOT_PASSWORD in .env}
      MYSQL_DATABASE: cbs_control
      MYSQL_USER: cbs_app
      MYSQL_PASSWORD: ${MYSQL_PASSWORD:?Set MYSQL_PASSWORD in .env}
    volumes:
      - mysql_data:/var/lib/mysql
    healthcheck:
      test: ["CMD-SHELL", "mysqladmin ping -h localhost -uroot -p\"$$MYSQL_ROOT_PASSWORD\""]
      interval: 10s
      timeout: 5s
      retries: 15
    networks:
      - cbs-internal

networks:
  cbs-internal:
    driver: bridge

volumes:
  mysql_data:
